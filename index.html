function apriModalPrenotazione(spaceId) { // 1. Controllo se l'utente è loggato if (!currentUser) { richiediLoginOAccedi('login'); return; } // 2. Recupero dati dello spazio const spz = localSpazi.find(s => s.id === spaceId); if (!spz || !spz.available) return; document.getElementById('bookSpaceId').value = spz.id; document.getElementById('bookBasePrice').value = spz.price; document.getElementById('bookSpaceTitle').innerText = 'Prenota: ' + spz.name; const daysStr = (spz.openDays || [1,2,3,4,5,6]).map(d => DAYS_MAP[d].substring(0,3)).join(', '); document.getElementById('bookOpeningHoursNotice').innerText = `🕒 Giorni aperti: ${daysStr} | Orario: ${spz.openTime || '09:00'} - ${spz.closeTime || '19:00'}`; const today = new Date().toISOString().split('T')[0]; document.getElementById('bookDate').value = today; document.getElementById('bookHourlyDate').value = today; document.getElementById('bookStartDate').value = today; document.getElementById('bookEndDate').value = today; setBookingMode('single'); toggleModal('bookingModal', true); }
